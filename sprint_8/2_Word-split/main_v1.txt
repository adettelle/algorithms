/*
-- ПРИНЦИП РАБОТЫ --
Считываем текст (строку без пробелов) и слова, которые формируют словарь.
Используем динамическое программирование. Заводим массив dp длины len(text)+1,
где dp[i] означает, что первые i символов текста можно полностью разбить на слова
из словаря.
dp[0] считаем true, потому что пустую строку можно считать уже корректно разбитой.
Далее последовательно рассматриваем все позиции i от 0 до len(text).
Если dp[i] == true, то первые i символов уже можно разбить на допустимые слова.
Поэтому попробуем поставить после них каждое слово из словаря.
Если это слово совпадает с текстом, начиная с позиции i,
то ставим dp[i + длина_слова] = true.
В конце проверяем последний элемент массива dp. Если он равен true, значит,
текст можно разбить на слова из данного словаря.
При этом одно и то же слово можно использовать любое количество раз,
поскольку при каждой позиции мы снова рассматриваем все слова словаря.

-- ДОКАЗАТЕЛЬСТВО КОРРЕКТНОСТИ --
Примем dp[0] = true, поскольку пустая строка не содержит символов и может считаться
разбитой на ноль слов.

Пусть dp[i] = true. Это означает, что первые i символов текста можно представить
как последовательность слов из словаря.
Алгоритм перебирает каждое слово word из словаря. Если word совпадает с частью текста,
начинающейся с позиции i, то к уже существующему разбиению первых i символов
можно добавить это слово.
В результате первые i + len(word) символов также представимы в виде последовательности
слов из словаря, поэтому алгоритм устанавливает dp[i + len(word)] = true.
Таким образом, каждое значение dp[i], устанавливаемое алгоритмом,
действительно означает, что соответствующий префикс текста можно разбить
на слова из словаря.

-- ВРЕМЕННАЯ СЛОЖНОСТЬ --
Пусть N - длина строки (текста), M - количество слов в словаре,
L - максимальная длина слова в словаре.
Для каждой из N позиций текста алгоритм в худшем случае перебирает все M слов словаря.
Проверка совпадения подстроки со словом может занимать O(L) времени.
Таким образом, общая временная сложность: O(N*M*L).
По условию M ≤ 100 и L ≤ 100, поэтому оба параметра ограничены константой.
Тогда итоговая временная сложность составляет O(N).

-- ПРОСТРАНСТВЕННАЯ СЛОЖНОСТЬ --
Пусть N - длина строки (текста), M - количество слов в словаре,
L - максимальная длина слова в словаре.
Массив dp содержит N + 1 элементов, поэтому занимает O(N) памяти.
Также хранится словарь из M слов. Если учитывать сами символы слов,
то для хранения словаря требуется O(M*L) памяти.
Итого общая пространственная сложность: O(N+M*L).
По условию M ≤ 100 и L ≤ 100, поэтому оба параметра ограничены константой.
Тогда итоговая временная сложность составляет O(N).
*/

package main

import (
	"bufio"
	"fmt"
	"os"
)

func main() {
	reader := bufio.NewReader(os.Stdin)
	var text string
	fmt.Fscan(reader, &text)

	var n int
	fmt.Fscan(reader, &n)

	dictionary := make([]string, 0, n)

	for i := 0; i < n; i++ {
		var word string
		fmt.Fscan(reader, &word)
		dictionary = append(dictionary, word)
	}

	res := IsCoveredByDictionary(text, dictionary)
	if !res {
		fmt.Println("NO")
	} else {
		fmt.Println("YES")
	}
}

func IsCoveredByDictionary(text string, dictionary []string) bool {
	dp := make([]bool, len(text)+1)
	dp[0] = true

	for i := 0; i < len(dp); i++ {
		if dp[i] == true {
			for _, word := range dictionary {
				if i+len(word) < len(dp) {
					if text[i:i+len(word)] == word {
						dp[i+len(word)] = true
					}
				}
			}
		}
	}

	return dp[len(dp)-1]
}
