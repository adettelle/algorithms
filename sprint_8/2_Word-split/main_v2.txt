/*
-- ПРИНЦИП РАБОТЫ --
Считываем текст (строку без пробелов) и слова, которые формируют словарь.
Для ускорения поиска слов строим префиксное дерево (Trie). Каждая вершина Trie
хранит переходы по буквам a-z и признак finish, который показывает,
заканчивается ли в этой вершине слово из словаря.
Также используем динамическое программирование. Заводим массив dp длины len(text)+1,
где dp[i] означает, что первые i символов текста можно полностью разбить на слова
из словаря.
dp[0] считаем true, потому что пустую строку можно считать уже корректно разбитой.
Далее последовательно рассматриваем все позиции i текста.
Если dp[i] == true, то первые i символов уже можно разбить на допустимые слова.
Из этой позиции начинаем двигаться по тексту вперёд и одновременно перемещаться
по Trie, начиная с корневой вершины. Для каждого следующего символа проверяем,
существует ли соответствующий переход в Trie.
Если перехода нет, дальнейшее движение из этой позиции невозможно,
поэтому прекращаем поиск. Если переход существует, переходим в следующую вершину Trie.
Если эта вершина имеет finish == true, значит, найдено слово из словаря,
которое начинается в позиции i и заканчивается в текущей позиции j.
В этом случае устанавливаем dp[j+1] = true.
В конце проверяем dp[len(text)]. Если оно равно true, значит, весь текст можно разбить
на слова из словаря.
Одно и то же слово можно использовать любое количество раз, поскольку из каждой
достижимой позиции мы снова начинаем поиск слов в Trie.

-- ДОКАЗАТЕЛЬСТВО КОРРЕКТНОСТИ --
Примем dp[0] = true, поскольку пустая строка не содержит символов и может считаться
разбитой на ноль слов.

Пусть dp[i] = true. Это означает, что первые i символов текста можно представить
как последовательность слов из словаря.

Алгоритм начинает из позиции i движение по тексту и одновременно проходит
соответствующий путь в Trie. Если при этом достигается вершина с finish == true,
значит, часть текста text[i:j+1] является словом из словаря.
Так как первые i символов уже можно разбить на слова, добавление найденного слова
позволяет разбить первые j+1 символов текста.
Поэтому алгоритм устанавливает dp[j+1] = true.
Таким образом, каждое значение dp[i], устанавливаемое алгоритмом,
действительно означает, что соответствующий префикс текста можно разбить
на слова из словаря.
Если в конце dp[len(text)] == true, это означает, что разбит весь текст.

-- ВРЕМЕННАЯ СЛОЖНОСТЬ --
Пусть N - длина строки (текста), M - количество слов в словаре,
L - максимальная длина слова в словаре.
Построение Trie требует времени O(M*L), так как каждый символ каждого слова
добавляется в Trie не более одного раза.
При обработке текста для каждой достижимой позиции i мы идём вперёд по тексту,
пока существует соответствующий путь в Trie. Поскольку длина любого слова не превышает
L, из одной позиции мы проверяем не более L символов.
Таким образом, обработка текста занимает O(N*L) в худшем случае.
Общая временная сложность: O(M*L + N*L).
По условию M ≤ 100 и L ≤ 100, поэтому эти параметры ограничены константой.
Следовательно, относительно длины текста N итоговая временная сложность
составляет O(N).

-- ПРОСТРАНСТВЕННАЯ СЛОЖНОСТЬ --
Пусть N - длина строки (текста), M - количество слов в словаре,
L - максимальная длина слова в словаре.
Массив dp содержит N + 1 элементов, поэтому занимает O(N) памяти.
Trie содержит не более M*L + 1 вершин, поскольку каждая новая вершина соответствует
одному символу, добавленному при построении Trie.
Для каждой вершины хранится массив из 26 переходов и признак finish,
поэтому память на Trie составляет O(M*L).
Итого общая пространственная сложность: O(N+M*L).
По условию M ≤ 100 и L ≤ 100, поэтому оба параметра ограничены константой.
Тогда итоговая временная сложность составляет O(N).
*/

package main

import (
	"bufio"
	"fmt"
	"os"
)

func main() {
	reader := bufio.NewReader(os.Stdin)
	var text string
	fmt.Fscan(reader, &text)

	var n int
	fmt.Fscan(reader, &n)

	dictionary := make([]string, 0, n)
	for i := 0; i < n; i++ {
		var word string
		fmt.Fscan(reader, &word)
		dictionary = append(dictionary, word)
	}

	res := IsCoveredByDictionary(text, dictionary)
	if !res {
		fmt.Println("NO")
	} else {
		fmt.Println("YES")
	}
}

type Trie struct {
	nodes []Node
}

type Node struct {
	next   [26]int
	finish bool
}

func NewNode() Node {
	node := Node{}
	for i := range node.next {
		node.next[i] = -1
	}
	return node
}

// i - позиция символа в слове
// index - номер символа в алфавите от 0 до 25
// v - номер текущей вершины trie
// номера Node присваиваются в порядке добавления слов
func BuildTrie(dictionary []string) *Trie {
	trie := &Trie{
		nodes: []Node{NewNode()},
	}

	for _, word := range dictionary {
		v := 0

		for i := 0; i < len(word); i++ {
			index := word[i] - 'a'

			if trie.nodes[v].next[index] == -1 {
				trie.nodes[v].next[index] = len(trie.nodes)
				trie.nodes = append(trie.nodes, NewNode())
			}
			v = trie.nodes[v].next[index]
		}
		trie.nodes[v].finish = true
	}

	return trie
}

func IsCoveredByDictionary(text string, dictionary []string) bool {
	trie := BuildTrie(dictionary)

	dp := make([]bool, len(text)+1)
	dp[0] = true

	for i := 0; i < len(text); i++ {
		if dp[i] == true {
			v := 0
			for j := i; j < len(text); j++ {
				index := text[j] - 'a'
				next := trie.nodes[v].next[index]
				if next == -1 {
					break
				}
				v = next
				if trie.nodes[v].finish {
					dp[j+1] = true
				}
			}
		}
	}
	return dp[len(text)]
}
